{{- /* whitney-hub — the data half of the Whitney 2026 series page. Everything here is derived from the `hike:` blocks in the series' posts (see CONVENTIONS.md), which in turn come from Basecamp's activities table. Nothing is hardcoded except Whitney itself, which hasn't been hiked yet and so has no post and no numbers to read. Sort order is `hike.hiked`, never `.Date`. Publish dates lag the hikes by 12-26 days and the Six-Pack wasn't even hiked in list order, so sorting by publish date puts the table in a sequence that never happened. */ -}} {{- /* ---- Gather: one row per post that carries a hike block ---- */ -}} {{- $rows := slice -}} {{- range $p := .Pages -}} {{- with $p.Params.hike -}} {{- $trackName := .track -}} {{- $trackUrl := "" -}} {{- with $p.Resources.GetMatch (default "" $trackName) -}} {{- $trackUrl = .RelPermalink -}} {{- end -}} {{- if and $trackName (not $trackUrl) -}} {{- errorf "whitney-hub: track %q not found in %s" $trackName $p.File.Path -}} {{- end -}} {{- $rows = $rows | append (dict "peak" .peak "elevation" (int .elevation_ft) "distance" (float .distance_mi) "gain" (int .gain_ft) "moving" .moving_time "hiked" (time.Format "2006-01-02" .hiked) "hikedLabel" (time.Format "Jan 2, 2006" .hiked) "sixpack" (default false .sixpack) "url" $p.RelPermalink "trackUrl" $trackUrl ) -}} {{- end -}} {{- end -}} {{- $rows = sort $rows "hiked" -}} {{- /* ---- Aggregate: summed over the rows above, never written by hand ---- */ -}} {{- $peakCount := len $rows -}} {{- $totalMiles := 0.0 -}} {{- $totalGain := 0 -}} {{- $highest := 0 -}} {{- range $rows -}} {{- $totalMiles = add $totalMiles .distance -}} {{- $totalGain = add $totalGain .gain -}} {{- if gt .elevation $highest -}}{{- $highest = .elevation -}}{{- end -}} {{- end -}}
